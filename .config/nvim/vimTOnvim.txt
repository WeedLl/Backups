Шаг 1. Установка Neovim
bash
# macOS (если используете Homebrew)
brew install neovim

# Или скачайте AppImage/binary с GitHub releases
# https://github.com/neovim/neovim/releases
Шаг 2. Структура конфигурации
Neovim использует XDG-директории. Ваш .vimrc переезжает в ~/.config/nvim/init.lua.

Рекомендуемая структура:

text
~/.config/nvim/
├── init.lua              # точка входа
├── lua/
│   ├── options.lua       # настройки редактора
│   ├── autocmds.lua      # автокоманды
│   ├── keymaps.lua       # горячие клавиши
│   └── plugins/
│       ├── init.lua      # lazy.nvim bootstrap + список плагинов
│       ├── lsp.lua       # LSP-конфигурация
│       ├── cmp.lua       # автодополнение
│       └── ...            # по одному файлу на крупный плагин

Что изменилось и почему:

set t_Co=256 — удалено. Neovim автоматически поддерживает 256 цветов и true color. Эта опция была нужна только в старых терминалах.
syntax on — удалено. В Neovim синтаксис включается автоматически. С Tree-sitter он заменяется на более точную подсветку.
let python_highlight_all = 1 — заменено на Tree-sitter.
set guifont=JetBrainsMono\ NFM:h14 — актуально только для GUI (Neovim-qt, Neovide). Если используете GUI, вынесите в отдельный конфиг GUI-фронтенда.
language en_US и set langmenu — в Neovim langmenu не релевантен (нет встроенного GUI-меню), но language messages C можно оставить, если нужны английские сообщения.
opt.foldcolumn = "2" — в Neovim это строковый параметр (может быть "auto", "auto:1" и т. д.).
~/.config/nvim/lua/autocmds.lua — автокоманды:

lua
local augroup = vim.api.nvim_create_augroup("UserAutocmds", { clear = true })

-- Переключение cursorline в режиме Insert
vim.api.nvim_create_autocmd({ "InsertEnter", "InsertLeave" }, {
  group = augroup,
  callback = function()
    vim.opt.cursorline = false
  end,
  pattern = "*",
})
-- Возвращаем cursorline при выходе из Insert
vim.api.nvim_create_autocmd("InsertLeave", {
  group = augroup,
  callback = function()
    vim.opt.cursorline = true
  end,
  pattern = "*",
})

-- Python: обрезка пробелов перед сохранением
vim.api.nvim_create_autocmd("BufWritePre", {
  group = augroup,
  pattern = "*.py",
  callback = function()
    local save = vim.fn.winsaveview()
    vim.cmd([[%s/\s\+$//e]])
    vim.fn.winrestview(save)
  end,
})

-- Python: умные отступы
vim.api.nvim_create_autocmd("BufRead", {
  group = augroup,
  pattern = "*.py",
  callback = function()
    vim.opt.smartindent = true
    vim.opt.cinwords = "if,elif,else,for,while,try,except,finally,def,class"
  end,
})

-- Закрытие Neovim, если последнее окно — файловое дерево
vim.api.nvim_create_autocmd("BufEnter", {
  group = augroup,
  pattern = "*",
  callback = function()
    local bufname = vim.api.nvim_buf_get_name(0)
    -- Работает для neo-tree и nvim-tree
    if vim.fn.tabpagenr("$") == 1 and vim.fn.winnr("$") == 1
       and (bufname:match("neo%-tree") or bufname:match("NvimTree")) then
      vim.cmd("quit")
    end
  end,
})
Что изменилось:

Вместо autocmd используется vim.api.nvim_create_autocmd — это Lua API, типобезопасный и с группами.
Ваш исходный autocmd для cursorline просто переключал cursorline! (toggle), что могло давать непредсказуемое поведение. Я разделил на чёткое «выключить в Insert, включить при выходе».
Обрезка пробелов: вместо normal m...я используюwinsaveview()/winrestview()` — надёжнее и не зависит от меток.
Автозакрытие при последнем окне: адаптировано под neo-tree/nvim-tree вместо NERDTree.
~/.config/nvim/lua/keymaps.lua — горячие клавиши:

lua
local map = vim.keymap.set

-- Folding
map("n", "<space>", "za", { silent = true, desc = "Toggle fold" })

-- Neo-tree toggle (вместо NERDTree)
map("n", "<C-n>", "<cmd>Neotree toggle<CR>", { silent = true, desc = "Toggle file tree" })

-- Перемещение по окнам с авто-созданием split
map("n", "<C-h>", function()
  local winnr = vim.fn.winnr()
  vim.cmd("wincmd h")
  if vim.fn.winnr() == winnr then
    vim.cmd("wincmd v")
    vim.cmd("wincmd h")
  end
end, { silent = true })

map("n", "<C-j>", function()
  local winnr = vim.fn.winnr()
  vim.cmd("wincmd j")
  if vim.fn.winnr() == winnr then
    vim.cmd("wincmd s")
    vim.cmd("wincmd j")
  end
end, { silent = true })

map("n", "<C-k>", function()
  local winnr = vim.fn.winnr()
  vim.cmd("wincmd k")
  if vim.fn.winnr() == winnr then
    vim.cmd("wincmd s")
    vim.cmd("wincmd k")
  end
end, { silent = true })

map("n", "<C-l>", function()
  local winnr = vim.fn.winnr()
  vim.cmd("wincmd l")
  if vim.fn.winnr() == winnr then
    vim.cmd("wincmd v")
    vim.cmd("wincmd l")
  end
end, { silent = true })
Что изменилось:

vim.keymap.set — современный API для маппингов. Поддерживает desc для which-key.nvim.
Ваша функция WinMove() переписана на Lua-замыкания. Логика та же: если после wincmd окно не сменилось, создаём split и переходим.
<C-n> теперь вызывает neo-tree вместо NERDTree.
<space> za — без изменений, просто в Lua-синтаксисе.

Шаг 5. Что произошло с YCM-специфичным конфигом
Ваш .vimrc содержал немало YCM-настроек. Вот как они переносятся:

YCM настройка	Что в Neovim
ycm_confirm_extra_conf = 0	Не нужна — LSP не требует extra conf
ycm_add_preview_to_completeopt	blink.cmp → documentation.auto_show = true
ycm_autoclose_preview_window_after_completion	blink.cmp автоматически закрывает
ycm_min_num_of_chars_for_completion = 2	blink.cmp работает с 1 символа, но можно настроить
ycm_collect_identifiers_from_comments_and_strings	LSP + buffer source делают это автоматически
ycm_enable_diagnostic_signs	vim.diagnostic.config({ signs = true })
ycm_enable_diagnostic_highlighting	vim.diagnostic.config({ underline = true })
ycm_always_populate_location_list	vim.diagnostic.setloclist на <leader>d
YcmDiags	vim.diagnostic.setloclist
YcmShowDetailedDiagnostic	vim.diagnostic.open_float
Функция YcmStatus() + таймер	LSP-статус в statusline через lualine.nvim или vim.lsp.get_clients()
--clang-completer	lspconfig.clangd.setup({}) (через mason)
Statusline: вместо вашей ручной YcmStatus() рекомендую поставить lualine.nvim — она умеет показывать LSP-статус, диагностику и многое другое без таймеров:

lua
-- добавить в plugins/init.lua:
{
  "nvim-lualine/lualine.nvim",
  dependencies = { "nvim-tree/nvim-web-devicons" },
  config = function()
    require("lualine").setup({
      sections = {
        lualine_a = { "mode" },
        lualine_b = { { "filename", path = 1 }, "modified", "readonly" },
        lualine_c = { "diagnostics" },
        lualine_x = { "encoding", "fileformat", "filetype" },
        lualine_y = { "progress" },
        lualine_z = { "location" },
      },
    })
  end,
},
Это заменяет ваш set statusline+=%{YcmStatus()}\ %f\ %m%r%h%w\ %=%l:%c\ %p%% и заодно set ruler.
