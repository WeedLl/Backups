    l - отобразить список папок в указанной директории.

    open - открыть (программу);

    #Ищет файлы по имени, типу, размеру и другим атрибутам файлов
    find (где) (как) (что);
    Примеры:
        - Найти все .log файлы в текущей директории и подпапках
        find . -name "*.log"

        - Найти только директории с определённым именем в текущей директории
        find . -type d -name "backup"

        - Найти файлы крупнее 10 МБ
        find . -type f -size +10M
        
        - Найти файлы, измененные за последние 24 часа
        find . -type f -mrime -1

        - Найти и сразу удалить файлы с расширением .tmp
        find . -name "*.tmp" -delete

        - Найти и сразу удалить файлы с расширением НЕ .tex
        find . -type f ! -name "*.tex" -delete # Поиск и удаление множества файлов
        . - поиск начинается в текущей директории
        -type f - искать только обычные файлы (не каталоги)
        ! -name "*.tex" - условие "не заканчивается на .tex
        -delete - сразу удалить найденные файлы

    #Ищет текст внутри файлов - по строкам, регулярным выражениям, шаблонам
    grep (что) (где);
    Примеры:
        - Найти строку "TODO" во всех файлах дериктории
        grep -r "TODO" .
        -r - рекурсивно по всем подпапкам

        - Найти "function" в script.sh, написать номер строки
        grep -rn "function" script.xh
        -n - номер строки

        - Найти "error" в system.log ,без учёта регистра
        grep -i "error" /var/log/system.log

        - Найти строки, в которых нет "comment" в file.txt
        grep -v "comment" file.txt
        -v - инверсия, выводит всё, кроме совпадений

        ?- Найти имена файлов с совпадениями (без самих строк)
        grep -rl "import" *.py
        -l - выводит только имена фалов, где есть совпадения

        find + grep
        - Найти все .py файлы, в которых есть слово "TODO"
        find . -name "*.py" -exec grep -l "TODO" {}\;
        find . -name "*.py" | xargs grep "TODO"

        -exec - передает имена файлов как аргументы в grep, через | он бы искал "TODO" в списке файлов/путей к файлам
        через | - выше написан второй вариант
        - exec будет запускать grep отдельно для каждого файла, а | xargs передаст сразу пачкой
        -exec - встроенная опция find, которая для каждого найденного файла запускает отдельный процесс указанной ком-ы
            -exec _com_name_ {} \;
                {} - плейсхолдер имен файлов
                \; - конец
                + вместо \; передаст все одной пачкой (аналог xargs)
        -xargs - берёт строки из стандартного вывода (stdin) и превращает их в аргументы другой команды

                           | -exec ... \;       | -exec ... +        | xargs
        Процесс на файл    | Да, отдельный      | Нет, один на все   | Нет, один на всех
        Скорость           | Медленнее          | Быстрее            | Быстрее
        Обработка пробелов | Безопасно          | Безопасно          | Нужно -print0
        Гибкость           | Только внутри find | Только внутри find | Работает с любым выводом
        
        Если в имени файлов есть пробелы - my file.log - xargs по умолчанию разобьёт на два аргумента my и file.log
        Правильное использование для перестраховки:
        find . -name "*log" -print0 | xargs -0 gzip
        -print0 - find разделяет имена нулевым байтом вместо перевода строки
        -0 - xargs принимает нулевой разделитель
        -exec такой проблемы не имеет - find сам подставляет имена целиком в {}

    
    less - отобразить вывод по одному экрану за раз.

        РЕЗЮМЕ УРОКА 1.4
            1. По приведённым ниже командам будет выполнено:
             CTRL-g - вывод информации о файле и текущей позиции каретки в этом файле
             SHIFT-G - переход на последнюю строку файла
             номер и SHIFT-G - переход к строке с указанным номером
             gg - переход на первую строку файла
            2. При вводе символа  /  с последующим набором слова, будет выполнен поиск
    	     этого слова ВПЕРЁД по тексту.
             При вводе символа  ?  с последующим набором слова, будет выполнен поиск
    	     этого слова НАЗАД по тексту.
             После показа первого совпадения, нажмите  n  для перехода к следующему
    	     слову в том же направлении поиска или SHIFT-N для поиска в
    	     противоположном направлении.
             При нажатии клавиш  CTRL-O  будет возврат к предыдущему слову, а при
    	     нажатии клавиш  CTRL-I  будет переход к ранее найденному слову.

             #/\c_слово_ принудительно отключает учитывание регистра
             #Если нажать * или #, то будет выполнен поиск слова, на котором стоит коретка вперед или назад соответ-
             #ственно, без учёта регистра.
             ctrl
             #f + _word_ - поиск буквы в строке. ; - перемещение к следующей букве, , - к предыдущей

            3. При нажатии  %  , когда каретка на одной из скобок ( ), [ ] или { },
    	     будет найдена её парная скобка.
            4. Чтобы заменить первое найденное слово в строке, наберите	:s/что/чем
             Чтобы заменить все найденные слова в строке, наберите	:s/что/чем/g
             Чтобы заменить в указанными интервале строк, наберите	:#,#s/что/чем/g
             Чтобы заменить все найденные слова в файле, наберите	:%s/что/чем/g
             Чтобы запрашивалось подтверждение, добавьте флаг 'c'	:%s/что/чем/gc
        РЕЗЮМЕ УРОКА 1.5
            1. По команде  :!command  будет исполнена указанная внешняя команда.
             Некоторые полезные примеры:
             (Windows)	(UNIX)
             :!dir		 :!ls	    - вывести перечень файлов в каталоге;
             :!del ФАЙЛ	 :!rm ФАЙЛ  - удалить файл с указанным наименованием.
            2. По команде  :w ФАЙЛ  , текущий редактируемый файл будет записан
             с указанным наименованием.
            3. Используя команды  v  , перемещение каретки и  :w ФАЙЛ  можно сохранить
             визуально выделенные строки в файл с указанным наименованием.
            4. По команде  :r ФАЙЛ  будет прочитан файл с указанным наименованием 
             и его содержимое помещено ниже позиции каретки.
            5. По команде  :r !dir  будет получен вывод команды dir и помещён ниже
             позиции каретки.
